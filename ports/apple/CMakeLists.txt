cmake_minimum_required(VERSION 3.22)
project(GridrunnerApple VERSION 1.0.0 LANGUAGES C CXX)

if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME MATCHES "^(Darwin|iOS)$")
    message(FATAL_ERROR "This optional build requires a macOS or iOS Apple toolchain. Use the source root for Windows/Android.")
endif()
if(NOT SDL3_SOURCE_DIR OR NOT EXISTS "${SDL3_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set SDL3_SOURCE_DIR to an SDL 3.2.24 source checkout.")
endif()
get_filename_component(SDL3_SOURCE_DIR "${SDL3_SOURCE_DIR}" ABSOLUTE)
set(GRPP_APPLE_PORT ON)
set(GRPP_BUILD_APP ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SDL_OPENGLES ON CACHE BOOL "" FORCE)
set(SDL_METAL ON CACHE BOOL "" FORCE)
set(GRPP_APPLE_BUNDLE_ID "org.openminotaur.gridrunnerplusplus" CACHE STRING "Apple application bundle identifier")
set(GRPP_APPLE_SIGNING_TEAM "" CACHE STRING "Apple development team for device signing")

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(GRPP_APPLE_IOS ON)
    set(GRPP_APPLE_REQUIRES_IOS true)
    message(WARNING "Experimental iOS startup configuration uses native GLES, not ANGLE/Metal. Device, rotation and safe-area validation remains required.")
else()
    set(GRPP_APPLE_REQUIRES_IOS false)
    set(GRPP_ANGLE_DIR "" CACHE PATH "Directory containing Metal-enabled libEGL.dylib and libGLESv2.dylib")
    foreach(lib IN ITEMS EGL GLESv2)
        if(NOT EXISTS "${GRPP_ANGLE_DIR}/lib${lib}.dylib")
            message(FATAL_ERROR "GRPP_ANGLE_DIR must contain lib${lib}.dylib built for the selected macOS architectures.")
        endif()
    endforeach()
    get_filename_component(GRPP_ANGLE_DIR "${GRPP_ANGLE_DIR}" ABSOLUTE)
endif()

add_subdirectory("../.." engine)
